# Creating customer table
CREATE TABLE CUSTOMER(
	CUSTOMER_CITY VARCHAR(50),
    	CUSTOMER_COUNTRY VARCHAR(50),
    	CUSTOMER_FNAME varchar(50),
    	CUSTOMER_ID VARCHAR(10) UNIQUE NOT NULL,
    	CUSTOMER_LNAME VARCHAR(50),
    	CUSTOMER_SEGMENT VARCHAR(50),
    	CUSTOMER_STATE VARCHAR(50),
    	CUSTOMER_STREET VARCHAR(50),
    	CUSTOMER_ZIPCODE VARCHAR(50),
    	primary key(CUSTOMER_ID)
    )

# Creating product table
CREATE TABLE PRODUCT(
	PRODUCT_CARD_ID VARCHAR(50) NOT NULL UNIQUE,
    PRODUCT_CATEGORY_ID VARCHAR(100),
    PRODUCT_IMAGE VARCHAR(300),
    PRODUCT_NAME VARCHAR(100),
    PRODUCT_PRICE FLOAT,
    CATEGORY_NAME VARCHAR(50),
    PRIMARY KEY(PRODUCT_CARD_ID)
    )

CREATE TABLE TRANSACTION(
	TYPE VARCHAR(50) NOT NULL,
    DAYS_FOR_SHIPPING_REAL INT NOT NULL,
    DAYS_FOR_SHIPMENT_SCHEDULED INT NOT NULL,
    BENEFIT_PER_ORDER FLOAT NOT NULL,
    SALES_PER_CUSTOMER FLOAT,
    DELIVERY_STATUS VARCHAR(50) NOT NULL,
    LATE_DELIVERY_RISK INT NOT NULL,
    MARKET VARCHAR(30) NOT NULL,
    ORDER_CITY VARCHAR(50) NOT NULL,
    ORDER_COUNTRY VARCHAR(50) NOT NULL,
    ORDER_CUSTOMER_ID VARCHAR(30) NOT NULL,
    ORDER_DATE_DATEORDERS DATETIME NOT NULL,
    ORDER_ID VARCHAR(50) NOT NULL,
    ORDER_ITEM_CARDPROD_ID VARCHAR(50) NOT NULL,
    ORDER_ITEM_DISCOUNT FLOAT NOT NULL,
    ORDER_ITEM_DISCOUNT_RATE NOT NULL,
    ORDER_ITEM_ID VARCHAR(50) NOT NULL,
    ORDER_ITEM_PRODUCT_PRICE FLOAT NOT NULL,
    ORDER_ITEM_PROFIT_RATIO FLOAT NOT NULL,
    ORDER_ITEM_QUANTITY INT NOT NULL,
    SALES FLOAT NOT NULL,
    ORDER_ITEM_TOTAL FLOAT,
    ORDER_PROFIT_PER_ORDER FLOAT,
    ORDER_REGION VARCHAR(50) NOT NULL,
    ORDER_STATE VARCHAR(50) NOT NULL,
    ORDER_STATUS VARCHAR(50) NOT NULL,
    ORDER_ZIPCODE VARCHAR(50) NOT NULL,
    PRODUCT_STATUS VARCHAR(50) NOT NULL,
    SHIPPING_DATE_DATEORDERS DATETIME NOT NULL,
    SHIPPING_MODE VARCHAR(30) NOT NULL,
    PRODUCT_CARD_ID VARCHAR(50) NOT NULL,
    CUSTOMER_ID VARCHAR(50) NOT NULL,

);

# Making inter relationships between transaction, customer and product tables
ALTER TABLE TRANSACTION
ADD CONSTRAINT FOREIGN KEY (CUSTOMER_ID) REFERENCES CUSTOMER(CUSTOMER_ID);

ALTER TABLE TRANSACTION
ADD CONSTRAINT FOREIGN KEY (PRODUCT_CARD_ID) REFERENCES PRODUCT(PRODUCT_CARD_ID);
